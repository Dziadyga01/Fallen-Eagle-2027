on_actions = {
	on_startup = {
		effect = { if = { limit = { country_exists = PMN } PMN = { set_variable = { EUR_zaleznosc = 65 } EUR_przelicz = yes add_dynamic_modifier = { modifier = EUR_administracja } set_country_flag = EK_eurokomisariat_tag } } if = { limit = { country_exists = SLS } SLS = { set_variable = { EUR_zaleznosc = 65 } EUR_przelicz = yes add_dynamic_modifier = { modifier = EUR_administracja } set_country_flag = EK_eurokomisariat_tag } } }
	}
	on_monthly_PMN = {
		effect = { if = { limit = { NOT = { has_global_flag = PL2027_prolog } }  if = { limit = { has_country_flag = EK_integracja } add_to_variable = { EUR_zaleznosc = 1 } clamp_variable = { var = EUR_zaleznosc min = 0 max = 100 } } if = { limit = { has_country_flag = EK_suwerennosc } add_to_variable = { EUR_zaleznosc = -2 } clamp_variable = { var = EUR_zaleznosc min = 0 max = 100 } } EUR_przelicz = yes  } }
	}
	on_monthly_SLS = {
		effect = { if = { limit = { NOT = { has_global_flag = PL2027_prolog } }  if = { limit = { has_country_flag = EK_integracja } add_to_variable = { EUR_zaleznosc = 1 } clamp_variable = { var = EUR_zaleznosc min = 0 max = 100 } } if = { limit = { has_country_flag = EK_suwerennosc } add_to_variable = { EUR_zaleznosc = -2 } clamp_variable = { var = EUR_zaleznosc min = 0 max = 100 } } EUR_przelicz = yes  } }
	}
}
