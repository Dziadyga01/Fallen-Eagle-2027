on_actions = {
	on_startup = {
		effect = { if = { limit = { country_exists = PSN } PSN = { set_variable = { EUR_zaleznosc = 70 } EUR_przelicz = yes add_dynamic_modifier = { modifier = EUR_administracja } } } }
	}
	on_monthly_PSN = {
		effect = { if = { limit = { NOT = { has_global_flag = PL2027_prolog } }  if = { limit = { has_country_flag = EUR_integracja } add_to_variable = { EUR_zaleznosc = 1 } clamp_variable = { var = EUR_zaleznosc min = 0 max = 100 } } if = { limit = { has_country_flag = EUR_samodzielnosc } add_to_variable = { EUR_zaleznosc = -2 } clamp_variable = { var = EUR_zaleznosc min = 0 max = 100 } } EUR_przelicz = yes if = { limit = { has_global_flag = TFE_faza_2 NOT = { has_country_flag = EUR_spor_cd } any_neighbor_country = { has_country_flag = PL2027_polska is_subject = no } } set_country_flag = { flag = EUR_spor_cd days = 180 } random_neighbor_country = { limit = { has_country_flag = PL2027_polska is_subject = no } country_event = { id = eur.2 days = 2 } } } if = { limit = { date > 2028.3.1 NOT = { has_country_flag = EUR_komisarz_cd } } set_country_flag = { flag = EUR_komisarz_cd days = 240 } country_event = { id = eur.1 days = 5 } }  } }
	}
}
