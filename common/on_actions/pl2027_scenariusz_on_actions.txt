		effect = {
			# 0.8.7: start 22.10.2027 – prolog jednej Rzeczypospolitej (zdarzenia prologu planuje PRL_start)

			# start 11.11.2027 – Polska podzielona, bez zadnego zdarzenia prologu
			else = {
				set_global_flag = PL2027_start_listopad
				random_list = { 50 = { set_global_flag = EK_los_pomorze } 50 = { } }
				if = { limit = { country_exists = POL POL = { is_ai = no } country_exists = SMR } SMR = { change_tag_from = POL } }
				if = { limit = { country_exists = KPR } KPR = { transfer_state = 745 set_capital = { state = 745 } } }
				PL2027_start_regionalny = yes
				EK_tusk_listopad = yes
			}
		}
	}
