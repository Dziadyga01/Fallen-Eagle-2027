ROS_start_wojny = {
	set_country_flag = ROS_operacja_start
	set_global_flag = ROS_wojna_mazurska
	set_variable = { ROS_dni_wojny = 0 }
	every_state = { limit = { OR = { is_owned_by = WMZ is_owned_by = MAZ is_owned_by = SUW } } set_state_flag = ROS_cel_mazury }
	if = { limit = { country_exists = WMZ } WMZ = { capital_scope = { set_state_flag = ROS_kluczowy } } }
	if = { limit = { country_exists = MAZ } MAZ = { capital_scope = { set_state_flag = ROS_kluczowy } } }
	if = { limit = { country_exists = SUW } SUW = { capital_scope = { set_state_flag = ROS_kluczowy } } }
	# 0.9.11: jeden fokus = jedno państwo; Sojusz Mazurski powstaje przed atakiem, więc sojusznicy bronią się razem z celem
	ROS_sojusz_mazurski = yes
	set_country_flag = ROS_inwazja_rozpoczeta
	if = { limit = { country_exists = WMZ WMZ = { is_subject = no } NOT = { has_war_with = WMZ } } declare_war_on = { target = WMZ type = annex_everything } }
	else_if = { limit = { country_exists = SUW SUW = { is_subject = no } NOT = { has_war_with = SUW } } declare_war_on = { target = SUW type = annex_everything } }
	else_if = { limit = { country_exists = MAZ MAZ = { is_subject = no } NOT = { has_war_with = MAZ } } declare_war_on = { target = MAZ type = annex_everything } }
	custom_effect_tooltip = WOS_tt_71
	add_ai_strategy = { type = conquer id = WMZ value = 200 }
	add_ai_strategy = { type = front_unit_request tag = WMZ value = 80 }
	add_ai_strategy = { type = conquer id = MAZ value = 200 }
	add_ai_strategy = { type = front_unit_request tag = MAZ value = 80 }
	add_ai_strategy = { type = conquer id = SUW value = 200 }
	add_ai_strategy = { type = front_unit_request tag = SUW value = 80 }
	ROS_oznacz_wartosc = yes
	capital_scope = { create_unit = { division = "division_template = \"Batalionowa Grupa Taktyczna\" start_experience_factor = 0.4" owner = ROS } create_unit = { division = "division_template = \"Batalionowa Grupa Taktyczna\" start_experience_factor = 0.4" owner = ROS } create_unit = { division = "division_template = \"Batalionowa Grupa Taktyczna\" start_experience_factor = 0.4" owner = ROS } create_unit = { division = "division_template = \"Batalionowa Grupa Taktyczna\" start_experience_factor = 0.4" owner = ROS } create_unit = { division = "division_template = \"Batalionowa Grupa Taktyczna\" start_experience_factor = 0.4" owner = ROS } create_unit = { division = "division_template = \"Batalionowa Grupa Taktyczna\" start_experience_factor = 0.4" owner = ROS } create_unit = { division = "division_template = \"Brygada Zmechanizowana\" start_experience_factor = 0.4" owner = ROS } create_unit = { division = "division_template = \"Brygada Zmechanizowana\" start_experience_factor = 0.4" owner = ROS } create_unit = { division = "division_template = \"Brygada Zmechanizowana\" start_experience_factor = 0.4" owner = ROS } create_unit = { division = "division_template = \"Brygada Artylerii\" start_experience_factor = 0.4" owner = ROS } create_unit = { division = "division_template = \"Brygada Artylerii\" start_experience_factor = 0.4" owner = ROS } }
	if = { limit = { country_exists = BLR } BLR = { give_military_access = ROS } }
	every_country = { limit = { OR = { tag = WMZ tag = MAZ tag = SUW } exists = yes } country_event = { id = ros.1 } }
	news_event = { id = ros.2 }
	if = { limit = { country_exists = ZKR } ZKR = { country_event = { id = zkr.10 days = 2 } } }
	ROS = { set_country_flag = ROS_interwencja set_country_flag = { flag = ROS_wojna_trwa days = 180 } set_variable = { ROS_cel_mazury = 0 } }
	every_state = { limit = { OR = { is_owned_by = WMZ is_owned_by = MAZ is_owned_by = SUW } } set_state_flag = ROS_cel_mazury ROS = { add_to_variable = { ROS_cel_mazury = 1 } } }
}
ROS_sojusz_mazurski = {
	if = { limit = { country_exists = WMZ }
		WMZ = {
			if = { limit = { is_in_faction = no } create_faction = PL2027_SOJUSZ_MAZURSKI }
			if = { limit = { country_exists = MAZ MAZ = { is_in_faction = no } is_faction_leader = yes } add_to_faction = MAZ }
			if = { limit = { country_exists = SUW SUW = { is_in_faction = no } is_faction_leader = yes } add_to_faction = SUW }
		}
	}
	else_if = { limit = { country_exists = SUW }
		SUW = {
			if = { limit = { is_in_faction = no } create_faction = PL2027_SOJUSZ_MAZURSKI }
			if = { limit = { country_exists = MAZ MAZ = { is_in_faction = no } is_faction_leader = yes } add_to_faction = MAZ }
		}
	}
}
