ZKR_wplywy_przelicz = {
	clamp_variable = { var = ZKR_wplywy min = 0 max = 100 }
	set_variable = { ZKR_wplywy_prod = ZKR_wplywy }
	multiply_variable = { ZKR_wplywy_prod = 0.002 }
	set_variable = { ZKR_wplywy_pp = ZKR_wplywy }
	multiply_variable = { ZKR_wplywy_pp = -0.003 }
	set_variable = { ZKR_wplywy_stab = ZKR_wplywy }
	multiply_variable = { ZKR_wplywy_stab = -0.001 }
	force_update_dynamic_modifier = yes
	if = {
		limit = { check_variable = { ZKR_wplywy > 99 } NOT = { has_country_flag = ZKR_protektorat } NOT = { has_country_flag = ZKR_wojna_domowa } NOT = { has_country_flag = ZKR_kryzys_100 } }
		set_country_flag = ZKR_kryzys_100
		country_event = { id = zkr.52 }
	}
}
ZKR_ideologia_namiestnika = {
	if = {
		limit = { country_exists = ROS ROS = { has_country_flag = ROS_lider_szo } }
		if = { limit = { NOT = { has_character = ZKR_radecki_w } } recruit_character = ZKR_radecki_w }
		set_politics = { ruling_party = nationalist elections_allowed = no }
		promote_character = ZKR_radecki_w
	}
}
ZKR_protektorat = {
	set_country_flag = ZKR_protektorat
	if = { limit = { country_exists = ROS } ROS = { puppet = ROOT } }
	if = { limit = { NOT = { has_character = ZKR_radecki } } recruit_character = ZKR_radecki }
	set_politics = { ruling_party = authoritarian_democrat elections_allowed = no }
	promote_character = ZKR_radecki
	ZKR_ideologia_namiestnika = yes
	set_cosmetic_tag = ZKR_NAMIESTNICTWO
	remove_dynamic_modifier = { modifier = ZKR_wplywy_mod }
	add_ideas = ZKR_namiestnictwo
	load_focus_tree = { tree = ZKR_rosyjskie_prusy_tree keep_completed = no }
}
ZKR_wojna_domowa = {
	set_country_flag = ZKR_wojna_domowa
	remove_dynamic_modifier = { modifier = ZKR_wplywy_mod }
	start_civil_war = { ideology = authoritarian_democrat size = 0.45 }
	random_country = {
		limit = { original_tag = ZKR NOT = { tag = ZKR } }
		set_country_flag = ZKR_rzad_tymczasowy
		set_cosmetic_tag = ZKR_RZAD_TYMCZASOWY
		recruit_character = ZKR_radecki
		promote_character = ZKR_radecki
		add_ideas = ZKR_wsparcie_rosji
		add_equipment_to_stockpile = { type = ranged_equipment amount = 2000 }
		if = { limit = { country_exists = ROS } ROS = { if = { limit = { PREV = { PL2027_ideologia_jak_PREV = yes } } give_guarantee = PREV } } }
		country_event = { id = zkr.71 }
	}
	country_event = { id = zkr.70 }
}
