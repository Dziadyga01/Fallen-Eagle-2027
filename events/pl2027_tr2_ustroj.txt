country_event = {
	id = tr2u.11
	title = tr2u.11.t
	desc = tr2u.11.d
	picture = GFX_pl2027_ev081_narada_2
	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = tr2u.11.a
		ai_chance = { base = 1 }
		country_event = { id = tr2end.101 days = 1 }
	}
	option = {
		name = tr2u.11.b
		ai_chance = { base = 2 modifier = { factor = 0 NOT = { country_exists = BYD } } }
		if = { limit = { country_exists = BYD NOT = { has_war_with = BYD } BYD = { is_subject = no } } BYD = { country_event = { id = tr2end.102 days = 2 } } } else = { country_event = { id = tr2end.103 days = 1 } }
	}
	option = {
		name = tr2u.11.c
		ai_chance = { base = 1 modifier = { factor = 2 has_war = yes } }
		set_country_flag = TR2END_gul_pact if = { limit = { is_in_faction = no } create_faction = TR2END_gul_pact }
		every_neighbor_country = { limit = { is_subject = no is_in_faction = no NOT = { has_war_with = ROOT } } country_event = { id = tr2end.104 days = 2 } }
	}
}
